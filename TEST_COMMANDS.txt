LIGHTING ROBUSTNESS - QUICK TEST REFERENCE
==========================================

BUILD COMMANDS
--------------

# Build only (verify compilation)
xcodebuild -scheme Tavi -destination 'platform=iOS Simulator,name=iPhone 15' build

# Clean build
xcodebuild -scheme Tavi -destination 'platform=iOS Simulator,name=iPhone 15' clean build


TEST COMMANDS
-------------

# Run all lighting robustness tests
xcodebuild test -scheme Tavi -destination 'platform=iOS Simulator,name=iPhone 15' \
  -only-testing:TaviTests/LightingRobustnessTests

# Run all tests (including existing + new)
xcodebuild test -scheme Tavi -destination 'platform=iOS Simulator,name=iPhone 15'

PATCH COMMANDS
--------------

# Check if patch applies cleanly
git apply --check lighting_robustness.patch

# Apply patch
git apply lighting_robustness.patch

# Show what patch will do (dry run)
git apply --stat lighting_robustness.patch

# Reverse patch (undo)
git apply -R lighting_robustness.patch


XCODE COMMANDS
--------------

# List available simulators
xcrun simctl list devices

# Open project in Xcode
open Tavi.xcodeproj

# Run tests from Xcode: Cmd+U


TROUBLESHOOTING
---------------

# If xcodebuild not found
sudo xcode-select -s /Applications/Xcode.app/Contents/Developer
sudo xcodebuild -license accept

# If simulator not found
xcrun simctl list devices
# Then replace 'iPhone 15' with available device name
